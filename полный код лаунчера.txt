def is_process_running(name):
    try:
        result = subprocess.run(
            f'tasklist /FI "IMAGENAME eq {name}" /NH',
            shell=True,
            capture_output=True,
            text=True
        )
        return name.lower() in result.stdout.lower()
    except:
        return False

def monitor_loop():
    global process_status
    while True:
        try:
            running = is_process_running("winws.exe") or is_process_running("goodbyedpi.exe")
            if running and process_status != "RUNNING":
                process_status = "RUNNING"
                root.after(0, update_ui_status, True)
            elif not running and process_status != "STOPPED":
                process_status = "STOPPED"
                root.after(0, update_ui_status, False)
            time.sleep(2)
        except:
            time.sleep(5)

process_status = "STOPPED"

def log_message(msg):

def find_file(folder, filename):
    if not os.path.exists(folder):
        return None
    for root, dirs, files in os.walk(folder):
        if filename in files:
            return os.path.join(root, filename)
    return None

def find_files(folder, ext):
    found = []
    if not os.path.exists(folder):
        return found
    for root, dirs, files in os.walk(folder):
        for f in files:
            if f.endswith(ext):
                found.append(os.path.join(root, f))
    return found

def auto_scan():
    global GOODBYE_EXE, WINDIVERT_DLL, WINDIVERT_SYS, BLACKLIST_FILE
    global ZAPRET_EXE, CYGWIN_DLL, BIN_FILES, BAT_FILES
    
    log_message("🔍 Поиск файлов...")
    
    goodbye_dir = os.path.join(BASE_DIR, "goodbyeDPI")
    if os.path.exists(goodbye_dir):
        GOODBYE_EXE = find_file(goodbye_dir, "goodbyedpi.exe") or GOODBYE_EXE
        WINDIVERT_DLL = find_file(goodbye_dir, "WinDivert.dll") or WINDIVERT_DLL
        WINDIVERT_SYS = find_file(goodbye_dir, "WinDivert64.sys") or WINDIVERT_SYS
        BLACKLIST_FILE = find_file(goodbye_dir, "russia-blacklist.txt") or BLACKLIST_FILE
    
    zapret_dir = os.path.join(BASE_DIR, "zapret")
    if os.path.exists(zapret_dir):
        ZAPRET_EXE = find_file(zapret_dir, "winws.exe") or ZAPRET_EXE
        CYGWIN_DLL = find_file(zapret_dir, "cygwin1.dll") or CYGWIN_DLL
    
    BIN_FILES = find_files(zapret_dir, '.bin') if os.path.exists(zapret_dir) else []
    
    # Ищем .bat файлы ВСЕГДА, не только в конкретных папках
    BAT_FILES = []
    for root, dirs, files in os.walk(BASE_DIR):
        for f in files:
            if f.endswith('.bat') or f.endswith('.cmd'):
                BAT_FILES.append(os.path.join(root, f))
    
    update_paths()
    update_config_list()  # <- Обновляем список конфигураций
    
    if GOODBYE_EXE:
        log_message(f"✅ GoodbyeDPI найден")
    if ZAPRET_EXE:
        log_message(f"✅ Zapret найден")
    if BAT_FILES:
        log_message(f"✅ Найдено .bat файлов: {len(BAT_FILES)}")

def load_config():
    global current_engine, current_arch, DNS_SERVER, SELECTED_CONFIG
    global GOODBYE_EXE, WINDIVERT_DLL, WINDIVERT_SYS, BLACKLIST_FILE
    global ZAPRET_EXE, CYGWIN_DLL
    
    try:
        if os.path.exists(CONFIG_FILE):
            with open(CONFIG_FILE, 'r', encoding='utf-8') as f:
                config = json.load(f)
                current_engine = config.get("engine", "GoodbyeDPI")
                current_arch = config.get("arch", "64")
                DNS_SERVER = config.get("dns", "8.8.8.8")
                SELECTED_CONFIG = config.get("config", "Стандартная")  # <- ВОТ ЭТО
                GOODBYE_EXE = config.get("goodbye_exe", "")
                WINDIVERT_DLL = config.get("windivert_dll", "")
                WINDIVERT_SYS = config.get("windivert_sys", "")
                BLACKLIST_FILE = config.get("blacklist_file", "")
                ZAPRET_EXE = config.get("zapret_exe", "")
                CYGWIN_DLL = config.get("cygwin_dll", "")
                return True
    except:
        pass
    return False

def save_config():
    try:
        config = {
            "engine": current_engine,
            "arch": current_arch,
            "dns": DNS_SERVER,
            "config": SELECTED_CONFIG,  # <- ВОТ ЭТО
            "goodbye_exe": GOODBYE_EXE,
            "windivert_dll": WINDIVERT_DLL,
            "windivert_sys": WINDIVERT_SYS,
            "blacklist_file": BLACKLIST_FILE,
            "zapret_exe": ZAPRET_EXE,
            "cygwin_dll": CYGWIN_DLL
        }
        with open(CONFIG_FILE, 'w', encoding='utf-8') as f:
            json.dump(config, f, indent=4, ensure_ascii=False)
        log_message("💾 Настройки сохранены")
        return True
    except:
        return False

def update_config_list():
    """Обновляет список доступных конфигураций в зависимости от движка"""
    configs = ["Стандартная"]
    
    if current_engine == "GoodbyeDPI":
        for f in BAT_FILES:
            name = os.path.basename(f).lower()
            if "goodbye" in name or "russia" in name or "dpi" in name:
                configs.append(os.path.basename(f))
    else:  # Zapret
        for f in BAT_FILES:
            name = os.path.basename(f).lower()
            if "zapret" in name or "alt" in name or "general" in name or "winws" in name:
                configs.append(os.path.basename(f))
    
    config_menu['values'] = configs
    
    # Если выбранная конфигурация есть в списке - оставляем
    if SELECTED_CONFIG in configs:
        config_var.set(SELECTED_CONFIG)
    else:
        config_var.set("Стандартная")

def run_zapret():
    try:
        log_message("🚀 Запуск Zapret...")
        
        if not ZAPRET_EXE or not os.path.exists(ZAPRET_EXE):
            log_message("❌ winws.exe не найден!")
            return
        
        # === ПРОВЕРЯЕМ .BAT КОНФИГУРАЦИЮ ===
        if SELECTED_CONFIG and SELECTED_CONFIG != "Стандартная":
            bat_path = None
            for f in BAT_FILES:
                if os.path.basename(f) == SELECTED_CONFIG:
                    bat_path = f
                    break
            
            if bat_path and os.path.exists(bat_path):
                log_message(f"📁 Использую .bat: {os.path.basename(bat_path)}")
                bat_dir = os.path.dirname(bat_path)
                
                subprocess.Popen(
                    os.path.basename(bat_path),
                    shell=True,
                    cwd=bat_dir,
                    creationflags=0
                )
                
                time.sleep(2)
                
                if is_process_running("winws.exe"):
                    log_message("✅ Zapret запущен через .bat!")
                    return
                else:
                    log_message("⚠️ .bat не сработал, пробуем прямой запуск")
        
        # === ПРЯМОЙ ЗАПУСК ===
        zapret_dir = os.path.dirname(ZAPRET_EXE)
        os.chdir(zapret_dir)
        
        if CYGWIN_DLL and os.path.exists(CYGWIN_DLL):
            shutil.copy2(CYGWIN_DLL, os.path.join(zapret_dir, "cygwin1.dll"))
        
        bin_files = find_files(zapret_dir, '.bin')
        tls = ""
        if bin_files:
            for f in bin_files:
                if "google" in f.lower() or "tls" in f.lower():
                    tls = f'--dpi-desync-fake-tls="{os.path.basename(f)}"'
                    break
            if not tls:
                tls = f'--dpi-desync-fake-tls="{os.path.basename(bin_files[0])}"'
        
        cmd = f'"{os.path.basename(ZAPRET_EXE)}" --wf-tcp=80,443 --wf-udp=443,443 --filter-udp=443 --dpi-desync=fake --dpi-desync-ttl=8 {tls} --dns-addr={DNS_SERVER}'
        
        log_message(f"📝 {cmd}")
        
        subprocess.Popen(cmd, shell=True, cwd=zapret_dir, creationflags=0)
        time.sleep(2)
        
    except Exception as e:
        log_message(f"❌ Ошибка: {e}")

def start_hero():
    if is_process_running("winws.exe") or is_process_running("goodbyedpi.exe"):
        messagebox.showwarning("Внимание", "Герой уже работает!")
        return
    
    if current_engine == "GoodbyeDPI":
        run_goodbyedpi()
    else:
        run_zapret()

def stop_hero():
    try:
        log_message("🛑 Остановка...")
        subprocess.run("taskkill /f /im goodbyedpi.exe", shell=True, capture_output=True)
        subprocess.run("taskkill /f /im winws.exe", shell=True, capture_output=True)
        log_message("✅ Остановлен")
    except Exception as e:
        log_message(f"❌ Ошибка: {e}")
